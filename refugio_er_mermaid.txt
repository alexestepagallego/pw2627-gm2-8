        int id PK
        varchar nombre
        varchar apellidos
        char dni UK
        date fecha_nacimiento
        varchar direccion
        date fecha_alta
    }
    administrativo {
        int id_trabajador PK, FK
    }
    monitor {
        int id_trabajador PK, FK
    }
    veterinario {
        int id_trabajador PK, FK
        varchar num_colegiado UK
    }
    disponibilidad_monitor {
        int id_monitor PK, FK
        enum dia_semana PK "lunes a viernes"
    }
    animal {
        char codigo_registro PK "15 caracteres"
        enum especie
        varchar raza "NULL"
        varchar nombre
        int edad
        enum estado_salud "sano / requiere_atencion_veterinaria"
        date fecha_llegada
        date fecha_fallecimiento "NULL"
        int id_recinto FK "NULL"
    }
    recinto {
        int id PK
        varchar nombre UK
        varchar ubicacion
        int capacidad_maxima
        enum tipo_animal
        int capacidad_actual "derivado, no se almacena"
    }
    historial_clinico {
        char codigo_animal PK, FK
        int id_veterinario FK
    }
    ingreso {
        int id PK
        char codigo_animal FK
        date fecha_entrada
        date fecha_salida "NULL"
        varchar medicacion "NULL"
    }
    adoptante {
        int id PK
        varchar nombre
        varchar apellidos
        char dni UK
        varchar direccion
        date fecha_nacimiento
        varchar telefono
    }
    adopcion {
        int id PK
        char codigo_animal FK, UK
        int id_adoptante FK
        int id_administrativo FK
        date fecha_adopcion
        boolean activa
    }
    centro_educativo {
        char codigo_centro PK "8 caracteres"
        varchar nombre
        varchar direccion
        varchar nombre_profesor
        varchar apellidos_profesor
        varchar email_profesor
    }
    solicitud_visita {
        int id PK
        char codigo_centro FK
        date fecha_solicitud
        int num_estudiantes
        enum nivel_educativo "infantil / primaria / secundaria"
        enum estado "pendiente / asignada / finalizada"
        int id_monitor FK "NULL"
        enum dia_asignado "NULL"
    }
    solicitud_dia_preferido {
        int id_solicitud PK, FK
        enum dia_semana PK "lunes a viernes"
    }
